Cara Deploy ke GitHub Pages (Agar Muncul Pop-up Install di Chrome)

  1. Buat repository baru di GitHub (misal: gas-pwa-app) dengan visibilitas Public.
  2. Buat file bernama index.html dan salin seluruh isi file ini ke dalamnya.
  3. Buat file kedua bernama sw.js (Service Worker) dengan kode yang tertera di bawah.
  4. Masuk ke menu repository: Settings > Pages.
  5. Pada bagian Branch, pilih main (atau root), lalu klik Save.
  6. Dalam 1-2 menit, link GitHub Pages Anda akan aktif (contoh: https://username.github.io/gas-pwa-app/). Buka di Chrome di Laptop/HP — tombol Install PWA akan otomatis muncul di browser bar!

const CACHE_NAME = 'gas-pwa-cache-v1';
const ASSETS = [
  './',
  './index.html',
  'https://cdn.tailwindcss.com',
  'https://cdn.jsdelivr.net/npm/bootstrap-icons@1.11.3/font/bootstrap-icons.min.css',
  'https://lh3.googleusercontent.com/d/1Uac4sUkGDifGEB1vD4mCB_r1dp4wftno'
];

self.addEventListener('install', (e) => {
  e.waitUntil(
    caches.open(CACHE_NAME).then((cache) => cache.addAll(ASSETS))
  );
  self.skipWaiting();
});

self.addEventListener('activate', (e) => {
  e.waitUntil(
    caches.keys().then((keys) => {
      return Promise.all(
        keys.map((key) => {
          if (key !== CACHE_NAME) return caches.delete(key);
        })
      );
    })
  );
  self.clients.claim();
});

self.addEventListener('fetch', (e) => {
  // Hanya intercept file lokal/static assets, iframe GAS tetap online
  if (e.request.url.includes('script.google.com')) {
    return;
  }
  e.respondWith(
    caches.match(e.request).then((res) => res || fetch(e.request))
  );
});

function setupDatabase() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  
  // Sheet Users
  var userSheet = ss.getSheetByName('Users') || ss.insertSheet('Users');
  if (userSheet.getLastRow() === 0) {
    userSheet.appendRow(['ID', 'Name', 'Email', 'Date']);
    userSheet.appendRow(['USR-101', 'Ahmad Rizky', 'ahmad@mail.com', new Date().toISOString().split('T')[0]]);
    userSheet.appendRow(['USR-102', 'Siti Rahma', 'siti@mail.com', new Date().toISOString().split('T')[0]]);
  }

  // Sheet Transactions
  var trxSheet = ss.getSheetByName('Transactions') || ss.insertSheet('Transactions');
  if (trxSheet.getLastRow() === 0) {
    trxSheet.appendRow(['ID', 'User', 'Amount', 'Date']);
    trxSheet.appendRow(['TRX-501', 'Ahmad Rizky', 150000, new Date().toISOString().split('T')[0]]);
    trxSheet.appendRow(['TRX-502', 'Siti Rahma', 250000, new Date().toISOString().split('T')[0]]);
  }

  return { status: 'success', message: 'Database setup berhasil!' };
}

function cleanData(sheetName) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName(sheetName);
  if (!sheet) return [];
  
  var values = sheet.getDataRange().getValues();
  if (values.length < 2) return [];
  
  var headers = values[0].map(function(h) { return h.toString().toLowerCase().trim(); });
  var result = [];
  
  for (var i = 1; i < values.length; i++) {
    var row = values[i];
    var obj = {};
    for (var j = 0; j < headers.length; j++) {
      var val = row[j];
      if (val instanceof Date) {
        val = Utilities.formatDate(val, Session.getScriptTimeZone(), 'yyyy-MM-dd');
      }
      obj[headers[j]] = val !== null ? val : '';
    }
    result.push(obj);
  }
  return result;
}

function getAllData() {
  return {
    users: cleanData('Users'),
    transactions: cleanData('Transactions'),
    config: { appName: 'Smart WebApp V4', version: '1.0.0' }
  };
}

function saveUser(data) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName('Users') || ss.insertSheet('Users');
  var newId = 'USR-' + Math.floor(1000 + Math.random() * 9000);
  var dateStr = data.date || Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'yyyy-MM-dd');
  
  sheet.appendRow([newId, data.name, data.email, dateStr]);
  return { status: 'success', id: newId };
}

function deleteUser(userId) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName('Users');
  if (!sheet) return { status: 'error', message: 'Sheet not found' };
  
  var values = sheet.getDataRange().getValues();
  for (var i = 1; i < values.length; i++) {
    if (String(values[i][0]) === String(userId)) {
      sheet.deleteRow(i + 1);
      return { status: 'success' };
    }
  }
  return { status: 'not_found' };
}

    let db = {
      users: [
        { id: "USR-1", name: "Ahmad Rizky", email: "ahmad@mail.com", date: "2026-01-01" },
        { id: "USR-2", name: "Budi Santoso", email: "budi@mail.com", date: "2026-01-02" },
        { id: "USR-3", name: "Siti Rahma", email: "siti@mail.com", date: "2026-01-03" }

    function startLoading() {
      Swal.fire({ title: 'Memuat...', allowOutsideClick: false, didOpen: () => Swal.showLoading() });
    }
    function endLoading() { Swal.close(); }

    function loadData() {
      startLoading();
      if (typeof google === 'undefined') {
        setTimeout(() => {
          renderUI();
          endLoading();
        }, 500);
      } else {
        google.script.run
          .withSuccessHandler(res => {
            db = res;
            renderUI();
            endLoading();
          })
          .withFailureHandler(err => {
            console.error(err);
            endLoading();
          })
          .getAllData();
      }
    }

    function renderUI() {
      const tbody = document.getElementById('user-rows');
      if (!db.users || db.users.length === 0) {
        tbody.innerHTML = '<tr><td colspan="5" class="text-center">Tidak ada data.</td></tr>';
        return;
      }
      tbody.innerHTML = db.users.map(u => `
        <tr>
          <td><span class="badge bg-secondary">${u.id}</span></td>
          <td class="fw-bold">${u.name}</td>
          <td>${u.email}</td>
          <td>${u.date}</td>
          <td>
            <button class="btn btn-sm btn-danger" onclick="deleteUserRow('${u.id}')">
              <i class="bi bi-trash"></i>
            </button>
          </td>
        </tr>
      `).join('');
    }

    function openAddModal() {
      Swal.fire({
        title: 'Tambah Pengguna Baru',
        html: `
          <input id="swal-name" class="swal2-input" placeholder="Nama Lengkap">
          <input id="swal-email" class="swal2-input" placeholder="Email" type="email">
        `,
        focusConfirm: false,
        showCancelButton: true,
        confirmButtonText: 'Simpan',
        preConfirm: () => {
          const name = document.getElementById('swal-name').value;
          const email = document.getElementById('swal-email').value;
          if (!name || !email) {
            Swal.showValidationMessage('Nama dan Email wajib diisi!');
            return false;
          }
          return { name, email, date: new Date().toISOString().split('T')[0] };
        }
      }).then(res => {
        if (res.isConfirmed) {
          saveUserData(res.value);
        }
      });
    }

    function saveUserData(newUser) {
      startLoading();
      if (typeof google === 'undefined') {
        newUser.id = 'USR-' + Math.floor(Math.random() * 9000 + 1000);
        db.users.push(newUser);
        renderUI();
        endLoading();
        Swal.fire('Sukses', 'Data berhasil ditambahkan (Simulasi)', 'success');
      } else {
        google.script.run
          .withSuccessHandler(() => {
            loadData();
            Swal.fire('Sukses', 'Data tersimpan ke Spreadsheet', 'success');
          })
          .saveUser(newUser);
      }
    }

    function deleteUserRow(id) {
      if (confirm('Hapus user ' + id + '?')) {
        startLoading();
        if (typeof google === 'undefined') {
          db.users = db.users.filter(u => u.id !== id);
          renderUI();
          endLoading();
        } else {
          google.script.run
            .withSuccessHandler(() => loadData())
            .deleteUser(id);
        }
      }
    }

    window.onload = loadData;
  </script>
